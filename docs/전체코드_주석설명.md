# 시오사쿠 전체 실행 코드 — 한국어 주석 설명

실행 코드 안의 `[학습]` 주석을 따라 읽으세요. 아래 코드는 파일별로 나뉜 실제 소스의 학습용 사본입니다. 실행은 프로젝트 폴더에서 `npm start`, 검증은 `npm run check`로 합니다.

기존 DB 구조·예측식·업무 동작은 유지했습니다. 별도 백엔드 서버는 아직 없습니다. 보관 원본과 최초 추출·주석 작성 도구는 앱 실행 코드가 아니므로 이 모음에서 제외했습니다.

```js
// [학습] 날짜 문자열과 요일을 다루는 공통 함수 모음입니다.


// [학습] YYYY-MM-DD 형식과 실제 달력에 존재하는 날짜인지를 함께 검사해 true/false를 반환합니다.
function isValidDay(date){
  // [학습] 정규식: 연도 4자리-월 2자리-일 2자리 형식을 검사합니다. ^와 $는 문자열 전체를 뜻합니다.
  if(!/^\d{4}-\d{2}-\d{2}$/.test(date||""))return false;
  // [학습] UTC 정오의 날짜를 만들고 다시 YYYY-MM-DD로 바꿔 원문과 비교합니다. 2월 30일처럼 자동 보정된 날짜를 거릅니다.
  const d=new Date(date+"T12:00:00Z");return !isNaN(d.getTime())&&d.toISOString().slice(0,10)===date;
}
// [학습] Date 객체를 사용자 기기의 현지 날짜 문자열로 만듭니다. padStart(2,"0")는 1을 01처럼 표시합니다.
const localDay=d=>`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`;
// [학습] 날짜의 요일 번호를 반환합니다. 일요일 0, 월요일 1, ... 토요일 6입니다.
const dayIndex=s=>new Date(`${s}T12:00:00`).getDay();
// [학습] 요일 번호가 [0,6]에 포함되면 주말입니다. 공휴일은 별도 반영하지 않는 기존 규칙입니다.
const weekend=s=>[0,6].includes(dayIndex(s));
```

```js
// [학습] 수량 문자열 해석과 숫자 유효성 검사. 표시·파싱·수동 입력 함수의 실패 처리 방식이 서로 다릅니다.


// [학습] 표시용 숫자 포맷: 1000을 1,000으로 만듭니다. 현재 || 0 때문에 null·빈값도 0으로 표시되는 점을 구분해 읽으세요.
const fmt=v=>Number(v||0).toLocaleString("ko-KR");
// [학습] JavaScript가 정확히 표현할 수 있는 안전한 정수인지, 0 이상인지 검사합니다. 음수·소수는 false입니다.
const validQty=v=>Number.isSafeInteger(v)&&v>=0;
// [학습] 카카오톡 수량 해석 함수입니다. raw는 원문, type은 stock(오픈) 또는 remaining(마감). 실패하면 null을 반환합니다.
function parseQty(raw,type){
  // [학습] 공백과 진열·보관 문구를 제거해 수량 해석에 사용할 문자열을 만듭니다.
  const str=String(raw??"").replace(/\s|진열|보관/g,"");
  // [학습] 마감의 품절·소진·완판을 수량 0으로 해석합니다. 원문의 상태는 별도 반환하지 않는 기존 동작입니다.
  if(type==="remaining"&&/^(품절|소진|완판)$/.test(str))return 0;
  // [학습] 미입고·입고없음도 0으로 바뀝니다. 명시적 미입고와 단순 수량 0이 합쳐지는 기존 한계가 있습니다.
  if(type==="stock"&&/^(미입고|입고없음)$/.test(str))return 0;
  // [학습] 숫자, 선택적인 개 단위, +로 연결한 숫자만 허용합니다. 예: 20, 20개, 40+20개.
  if(!/^\d+(?:개)?(?:\+\d+(?:개)?)*$/.test(str))return null;
  // [학습] 개를 지우고 +로 나눈 뒤 각 조각을 Number로 숫자화합니다. "40+20개" → [40,20].
  const parts=str.replace(/개/g,"").split("+").map(Number);
  // [학습] reduce는 배열을 누적 계산합니다. 초기값 0에서 각 숫자를 더해 총수량을 구합니다.
  const sum=parts.reduce((s,v)=>s+v,0);
  // [학습] 합계도 안전한 정수인지 검사한 후 반환합니다.
  return validQty(sum)?sum:null;
}
// [학습] 수동 입력용입니다. 공란은 null, 올바른 숫자는 정수, 잘못된 값은 throw로 오류를 발생시킵니다.
function optionalQty(raw){
  // [학습] 미기록은 0과 구분해 null로 표현합니다.
  if(raw===""||raw===undefined||raw===null)return null;
  // [학습] 앞뒤 공백을 제거하고 숫자로만 구성됐는지 확인합니다. throw된 오류는 화면의 try/catch가 처리합니다.
  const str=String(raw).trim();if(!/^\d+$/.test(str))throw new Error(`수량은 0 이상의 정수만 허용합니다: ${str}`);
  // [학습] 문자열을 숫자로 바꾼 뒤 안전한 범위인지 다시 확인합니다.
  const num=Number(str);if(!validQty(num))throw new Error("수량이 허용 범위를 초과했습니다.");
  // [학습] 검증을 통과한 숫자를 호출한 쪽에 돌려줍니다.
  return num;
}

// [학습] 이 파일의 함수를 외부에서 import할 수 있도록 공개합니다.
export { fmt, validQty, parseQty, optionalQty };
```

```js
  if(branch==="강동 본점"){
    // [학습] has는 키 존재 여부입니다. 사전에 없는 이름은 임의로 새 상품으로 확정하지 않습니다.
    if(!gangdongMap.has(key))return {name:null,reason:`표준화 미확정: ${raw}`};
    const name=gangdongMap.get(key);
    // [학습] 삼항 연산자: name이 있으면 {name}, 없으면 제외 사유를 반환합니다. {name}은 {name:name}의 축약입니다.
    return name?{name}:{name:null,reason:"확정 제외 상품"};
  }
  // [학습] 다른 지점에서는 먼저 공통 별칭을 확인합니다.
  if(commonMap.has(key)){
    const name=commonMap.get(key);
    // [학습] some은 조건에 맞는 상품이 하나라도 있는지 확인합니다. !로 뒤집어 기존 목록에 없음을 표시합니다.
    return {name,newProduct:!products.some(p=>p.name===name)};
  }
  // [학습] 공통 별칭에 없으면 DB에서 받은 각 상품의 이름과 별칭을 차례로 비교합니다.
  for(const p of products){
    // [학습] 표준 이름 자체가 입력과 일치하면 해당 상품을 반환합니다.
    if(keyName(p.name)===key)return {name:p.name};
    // [학습] aliases가 없으면 []를 사용합니다. 별칭 중 하나라도 일치하면 그 상품 이름을 반환합니다.
    if((p.aliases||[]).some(alias=>keyName(alias)===key))return {name:p.name};
  }
  // [학습] 어느 규칙에도 없으면 정리한 원문을 신규 상품 후보로 반환합니다. 검증된 정식 상품이라는 뜻은 아닙니다.
  return {name:cleanName(raw),newProduct:true};
}
```

```js
  let y,m,d;
  // [학습] 숫자 부분만 뽑아 Number로 변환하고 [y,m,d]로 나눠 대입합니다.
  if(ymd){const nums=ymd[0].match(/\d+/g).map(Number);[y,m,d]=nums;}
  else{
    // [학습] 연도가 없으면 11월 1일 형태의 월·일을 찾습니다.
    const kor=String(text).match(/(\d{1,2})\s*월\s*(\d{1,2})\s*일/);
    // [학습] 11/1, 11-1 등 짧은 형식도 찾습니다.
    const short=String(text).match(/(?:^|\s)(\d{1,2})[.\/\-\s]+(\d{1,2})(?:\s|$)/);
    // [학습] 한글 월·일을 찾았을 때 정규식 캡처 그룹 1·2를 사용합니다.
    if(kor){m=Number(kor[1]);d=Number(kor[2]);}

    else if(short){m=Number(short[1]);d=Number(short[2]);}
    // [학습] 날짜를 찾지 못하면 화면에서 고른 날짜를 쓰고 추정 사용 여부를 표시합니다.
    else return {date:fallback,usedFallback:true};
    // [학습] 월·일만 있으면 대체 날짜의 연도를 사용합니다. 원문에 연도가 있었다는 뜻은 아닙니다.
    y=Number(String(fallback).slice(0,4));

  const date=`${y}-${String(m).padStart(2,"0")}-${String(d).padStart(2,"0")}`;

  const headers=lines.filter(line=>!line.includes("/")&&/(오픈|마감|입고|17\s*시|17:00|오후\s*5\s*시)/.test(line));

    // [학습] 지점과 상품 사전을 사용해 표준 이름을 찾습니다.
    const found=normalizeProduct(match[1],branch,products);
    // [학습] 표준화 실패는 오류 목록에 남긴 뒤 다음 줄로 넘어갑니다.
    if(!found.name){issues.push(`${match[1]}: ${found.reason}`);continue;}
    // [학습] 40+20개, 품절 등 수량 표현을 해석합니다.
    const qty=parseQty(match[2],type);
    // [학습] 해석 불가 수량은 원문과 함께 오류로 남깁니다.
    if(qty===null){issues.push(`${match[1]}: 수량 형식 오류 (${match[2]})`);continue;}
    // [학습] 여러 원문이 같은 표준 상품으로 모이면 임의 합산하지 않고 중복 오류로 처리합니다.
    if(Object.prototype.hasOwnProperty.call(items,found.name)){
      issues.push(`${found.name}: 같은 보고에 중복 상품이 있어 검토가 필요합니다.`);continue;
    }
    // [학습] 정상 상품 수량을 결과에 추가합니다.
    items[found.name]=qty;
    // [학습] 기존 상품 목록과 신규 후보 목록 모두에 없을 때만 새 후보를 추가합니다.
    if(!products.some(p=>p.name===found.name)&&!newProducts.includes(found.name))newProducts.push(found.name);
  }
  // [학습] 최종 상품 객체가 비어 있으면 유효 수량 없음 오류를 추가합니다.
  if(!Object.keys(items).length)issues.push("'상품명 / 20개' 형식의 유효한 수량이 없습니다.");
  // [학습] 해석 결과 전체를 화면에 돌려줍니다. branch는 현재 반환값에 없고 저장 호출부에서 추가합니다.
  return {date,type,items,issues,newProducts,estimated,usedFallback};
}
```

```js
// [학습] Excel 업로드 → 행 배열 → 날짜별 등록 후보를 생성합니다.
// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.

// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.

// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.

// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.

// [학습] Excel 날짜를 YYYY-MM-DD로 통일합니다. 입력은 날짜 일련번호·Date 객체·문자열일 수 있습니다.
function excelDate(value){
  // [학습] Excel 숫자 날짜는 SheetJS로 연·월·일로 해석합니다. 숫자 경로는 아래 문자열 경로와 검증 방식이 다릅니다.
  if(typeof value==="number"&&Number.isFinite(value)){
    // [학습] Excel 날짜 해석 도구를 호출합니다. window는 브라우저의 전역 객체입니다.
    const d=window.XLSX.SSF.parse_date_code(value);
    // [학습] 해석 실패는 null, 성공은 0을 채운 월·일 문자열로 반환합니다.
    if(!d)return null;return `${d.y}-${String(d.m).padStart(2,"0")}-${String(d.d).padStart(2,"0")}`;
  }
  // [학습] Date 객체면 공통 localDay 함수를 사용합니다.
  if(value instanceof Date)return localDay(value);
  // [학습] 문자열 입력을 준비합니다. null·undefined는 빈 문자열로 바꿉니다.
  const str=String(value??"").trim();
  // [학습] 연도를 포함하는 날짜 형식을 찾습니다.
  const m=str.match(/^(20\d{2})[-./년]\s*(\d{1,2})[-./월]\s*(\d{1,2})/);
  // [학습] 날짜 형식이 아니면 null로 표시합니다.
  if(!m)return null;
  // [학습] 정규식이 추출한 연·월·일을 표준 문자열로 조합합니다.
  const d=`${m[1]}-${m[2].padStart(2,"0")}-${m[3].padStart(2,"0")}`;
  // [학습] 문자열 경로에서는 실제 달력 날짜인지도 검사합니다.
  return isValidDay(d)?d:null;
}
// [학습] 빈 셀은 null, 잘못된 수량은 undefined, 정상 수량은 숫자로 반환하는 현재 계약입니다.
function excelQty(value,type){
  // [학습] 빈 셀과 숫자 0을 구분합니다.
  if(value===null||value===undefined||String(value).trim()==="")return null;
  // [학습] 마감 품절 문구는 0, 오픈의 품절 문구는 오류(undefined)입니다.
  if(typeof value==="string"&&/^(품절|소진|완판)$/.test(value.trim()))return type==="remaining"?0:undefined;
  // [학습] Excel 숫자 날짜는 SheetJS로 연·월·일로 해석합니다. 숫자 경로는 아래 문자열 경로와 검증 방식이 다릅니다.
  if(typeof value==="number")return validQty(value)?value:undefined;
  // [학습] 문자열 수량의 천 단위 쉼표를 제거합니다.
  const text=String(value).trim().replace(/,/g,"");
  // [학습] 공통 수량 파서로 개 단위·합산 등의 표현을 처리합니다.
  const parsed=parseQty(text,type);
  // [학습] 공통 파서의 실패 null을 Excel 처리의 오류 값 undefined로 맞춥니다.
  return parsed===null?undefined:parsed;
}
// 날짜 중 하나라도 중복·알 수 없는 상품·잘못된 수량이 있으면 그 날짜 전체를 격리하여 부분 등록 방지.
// [학습] 행 배열, 등록 지점·종료일, 기존 기록·상품 목록을 받아 등록 후보와 검토 결과를 만듭니다. DB에는 쓰지 않습니다.
function analyzeExcelRows(rows,branch,cutoff,existing,products){
  // [학습] 날짜·상품명 셀이 포함된 첫 행을 헤더로 찾습니다. 배열 인덱스는 0부터 시작합니다.
  const headerIndex=rows.findIndex(row=>Array.isArray(row)&&row.includes("날짜")&&row.includes("상품명"));
  // [학습] findIndex 결과 -1은 헤더를 못 찾았다는 뜻입니다.
  if(headerIndex<0)throw new Error("'날짜', '상품명', '오픈재고', '잔여재고' 열을 확인하세요.");
  // [학습] 헤더 셀의 앞뒤 공백을 제거합니다.
  const h=rows[headerIndex].map(x=>String(x??"").trim());
  // [학습] 필수 열 이름이 실제 몇 번째 열에 있는지 기록합니다. 열 순서가 바뀌어도 찾을 수 있습니다.
  const idx=["날짜","상품명","오픈재고","잔여재고"].map(v=>h.indexOf(v));
  // [학습] 필수 열 중 하나라도 없으면 분석을 중단합니다.
  if(idx.some(i=>i<0))throw new Error("필수 열이 누락됐습니다: 날짜/상품명/오픈재고/잔여재고");
  // [학습] days는 날짜별 상품 Map, blocked는 등록 제외 날짜 Set, warnings는 사용자 검토 문구입니다.
  const days=new Map(),blocked=new Set(),warnings=[];
  // [학습] 미기록·중복·범위 밖·수량 오류 등의 처리 건수를 누적합니다. 최종 등록 건수와 같지는 않습니다.
  const counts={source:0,missingOpen:0,missingClose:0,zeroOpen:0,stockout:0,invalid:0,duplicates:0,excludedItems:0,outside:0,badStock:0};

  // [학습] 오류 문구와 건수를 기록하고 날짜를 알 수 있을 때 해당 날짜 전체를 제외합니다. 날짜 자체가 잘못되면 특정 날짜를 격리할 수 없습니다.
  const note=(day,msg)=>{warnings.push(msg);counts.invalid++;if(day)blocked.add(day);};
  // [학습] 헤더 다음 행부터 검사합니다. forEach 콜백의 return은 해당 행만 건너뜁니다.
  rows.slice(headerIndex+1).forEach((row,i)=>{
    // [학습] 사용자에게 보여줄 Excel 행 번호는 1부터 세므로 인덱스를 보정합니다.
    const rowNo=headerIndex+2+i;
    // [학습] 빈 행은 처리하지 않습니다. every는 모든 셀이 빈값인지 검사합니다.
    if(!Array.isArray(row)||row.every(v=>v===null||v===undefined||v===""))return;
    // [학습] 날짜와 상품명을 각각 해석합니다.
    const day=excelDate(row[idx[0]]),raw=cleanName(row[idx[1]]);
    // [학습] 날짜·상품명 오류는 검토 목록에 남깁니다.
    if(!day||!raw){note(day,`${rowNo}행: 날짜/상품명 오류`);return;}
    // [학습] 종료일 이후와 강동점 시작일 이전은 범위 밖으로 건너뜁니다. 표준 날짜 문자열은 사전순으로 날짜 비교가 가능합니다.
    if(day>cutoff||(branch==="강동 본점"&&day<GANGDONG_MIN_DATE)){counts.outside++;return;}
    // [학습] 상품명을 지점별 규칙으로 표준화합니다.
    const mapped=normalizeProduct(raw,branch,products);
    // [학습] 표준화 불가와 확정 제외를 나눠 처리합니다.
    if(!mapped.name){
      // [학습] 확정 제외 상품은 행만 빼고 날짜 전체는 막지 않습니다.
      if(mapped.reason==="확정 제외 상품"){counts.excludedItems++;return;}
      // [학습] 미확정 상품은 날짜 전체를 막고 다음 행으로 넘어갑니다.
      note(day,`${rowNo}행: ${mapped.reason}`);return;
    }
    // [학습] 오픈·마감 수량을 각각 해석합니다.
    const stock=excelQty(row[idx[2]],"stock"),remaining=excelQty(row[idx[3]],"remaining");
    // [학습] 잘못된 수량이면 날짜 전체를 격리합니다. 공란 null은 여기서 오류로 취급하지 않습니다.
    if(stock===undefined||remaining===undefined){counts.badStock++;note(day,`${rowNo}행: 수량 형식 오류 (${raw})`);return;}
    // [학습] 처음 본 날짜면 해당 날짜의 상품 Map을 만듭니다.
    if(!days.has(day))days.set(day,new Map());
    // [학습] 현재 날짜에 모은 상품 목록을 가져옵니다.
    const items=days.get(day);
    // [학습] 같은 날짜에 표준 이름이 중복되면 합산하지 않고 날짜를 격리합니다.
    if(items.has(mapped.name)){counts.duplicates++;note(day,`${rowNo}행: ${mapped.name} 상품명이 중복 표준화됨`);return;}
    // [학습] 정상 형식의 재고와 파생 상태를 날짜별 Map에 추가합니다.
    items.set(mapped.name,{stock,remaining,no_delivery:stock===0,stockout:validQty(stock)&&stock>0&&remaining===0,missing_open:stock===null,missing_close:remaining===null});
    // [학습] 처리된 수량 및 누락 건수를 올립니다. 이후 날짜가 격리되면 이 값과 실제 등록 결과는 다를 수 있습니다.
    counts.source++;if(stock===null)counts.missingOpen++;if(remaining===null)counts.missingClose++;
    // [학습] 오픈 0·마감 0 관측의 건수를 셉니다.
    if(stock===0)counts.zeroOpen++;if(stock>0&&remaining===0)counts.stockout++;
    // [학습] 마감>오픈은 경고를 남기지만 현재 코드는 등록을 막지 않습니다. 집계 단계에서 제외합니다.
    if(stock!==null&&remaining!==null&&remaining>stock)warnings.push(`${rowNo}행: ${mapped.name} 마감 > 오픈 (분석 제외)`);

  });
  // [학습] 날짜와 지점을 합쳐 기존 기록을 찾는 키를 만듭니다. Set으로 중복 날짜 등록을 판별합니다.
  const existingKeys=new Set(existing.map(r=>`${r.date}|||${r.branch}`));
  // [학습] 신규 기록과 기존이라 건너뛴 날짜를 나눠 담습니다.
  const newRecords=[],skipped=[];
  // [학습] 날짜순으로 정렬해 등록 후보를 구성합니다.
  for(const [day,items] of [...days.entries()].sort(([a],[b])=>a.localeCompare(b))){
    // [학습] 품질 오류로 막힌 날짜는 등록 후보에 넣지 않습니다.
    if(blocked.has(day))continue;
    // [학습] Map을 일반 객체로 바꾸고 Excel 출처·마감 관련 메타데이터를 붙입니다.
    const row={date:day,branch,items:Object.fromEntries(items),remaining_source:"엑셀 일괄등록",remaining_time:null,is_estimated_close:false};
    // [학습] 같은 지점·날짜가 있으면 덮어쓰지 않습니다.
    if(existingKeys.has(`${day}|||${branch}`))skipped.push(day);else newRecords.push(row);
  }
  // [학습] 신규 기록 안의 상품명을 펼치고 Set으로 중복을 제거합니다.
  const productNames=[...new Set(newRecords.flatMap(record=>Object.keys(record.items)))];
  // [학습] 화면 미리보기와 실제 등록에 필요한 후보·경고·집계를 반환합니다.
  return {newRecords,skipped,blocked:[...blocked],counts,warnings,productNames};
}

// [학습] 파일 읽기는 시간이 걸리므로 async/await를 씁니다. 이 함수도 실제 DB 저장은 하지 않습니다.
async function readExcel(file,branch,cutoff,records,products){
  // [학습] CDN 라이브러리가 로드되지 않은 상황을 오류로 알립니다.
  if(!window.XLSX)throw new Error("엑셀 라이브러리를 불러오지 못했습니다.");
  // [학습] 사용자가 선택한 파일을 바이너리 데이터 ArrayBuffer로 읽습니다. await는 완료까지 이 함수의 다음 처리를 기다립니다.
  const buffer=await file.arrayBuffer();
  // [학습] SheetJS로 Excel 문서를 해석합니다. cellDates:false이므로 날짜를 주로 숫자로 받습니다.
  const workbook=window.XLSX.read(buffer,{type:"array",cellDates:false});
  // [학습] 재고 데이터 이름의 시트를 우선 사용하고 없으면 첫 시트를 사용합니다.
  const sheet=workbook.Sheets["재고 데이터"]||workbook.Sheets[workbook.SheetNames[0]];
  // [학습] 사용할 시트가 없으면 오류를 발생시킵니다.
  if(!sheet)throw new Error("엑셀 시트가 없습니다.");
  // [학습] header:1은 행 배열, defval:null은 빈 셀 표현, raw:true는 원시값 사용입니다.
  const rows=window.XLSX.utils.sheet_to_json(sheet,{header:1,defval:null,raw:true});
  // [학습] 행 분석 결과에 파일명·등록 조건을 추가합니다. 나중에 조건이 바뀌었는지 확인할 때 사용합니다.
  return {...analyzeExcelRows(rows,branch,cutoff,records,products),filename:file.name,branch,cutoff};
}
```

```js
  // [학습] 500건씩 반복 조회합니다. 종료 조건은 아래에서 결과 수로 판단합니다.

      .in("branch",BRANCHES).order("record_date").order("branch").range(start,start+499);

  return all.map(r=>({date:r.record_date,branch:r.branch,items:r.items||{},remaining_source:r.remaining_source,

async function saveRecord(record,expected){
  // [학습] 지점·날짜를 확인합니다. 현재 브라우저 검증이며 서버/DB에서의 권한 검사를 대신하지 않습니다.

  const payload={items:record.items,remaining_source:record.remaining_source||null,
    // [학습] 마감 시각과 대체 여부를 저장 형식으로 맞춥니다.

    // [학습] DB 오류를 호출한 함수의 catch에서 처리하도록 전달합니다.
    if(error)throw error;
  }else{
    // [학습] 수정할 행을 날짜와 지점으로 좁힙니다. let은 조건 추가로 request를 다시 대입하기 때문에 사용합니다.
    let request=db.from("inventory_records").update(payload).eq("record_date",record.date).eq("branch",record.branch);

    request=expected.updated_at?request.eq("updated_at",expected.updated_at):request.is("updated_at",null);

    if(!data?.length)throw new Error("다른 사용자가 먼저 수정했거나 수정 권한이 없습니다. 새로고침 후 다시 확인하세요.");
  }
```

```js
// [학습] 정상 관측을 집계하는 분석 함수입니다. 판매 수량을 계산하며 가격 기반 매출은 계산하지 않습니다.
// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.

// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.
import { canonicalRecord, stockStatus, observedSales } from './inventory.js';

// [학습] 여러 재고 기록을 상품별·지점별 판매량과 데이터 품질 통계로 바꿉니다. DB 조회는 하지 않습니다.
function summarize(records,products=[]){
  // [학습] 상품·지점별 누적값은 Map에, 품질별 건수는 quality 객체에 보관합니다.
  const byProduct=new Map(),byBranch=new Map(),quality={normal:0,missingOpen:0,missingClose:0,invalid:0,estimated:0,noDelivery:0,unmapped:0,collisions:0};
  // [학습] 날짜·지점별 기록을 하나씩 순회합니다.
  for(const raw of records){
    // [학습] 이름 표준화와 충돌 처리를 적용한 분석용 기록을 만듭니다.
    const r=canonicalRecord(raw,products);
    // [학습] 표준화 제외 건수와 상품명 충돌 건수를 전체 통계에 더합니다.
    quality.unmapped+=r.quality.skipped;quality.collisions+=r.quality.collisions;
    // [학습] 처음 본 지점이면 누적값을 준비합니다. Set은 같은 날짜를 중복 집계하지 않습니다.
    if(!byBranch.has(r.branch))byBranch.set(r.branch,{days:new Set(),sales:0,remaining:0,valid:0});
    // [학습] 현재 지점의 누적 객체를 가져와 보고 날짜를 추가합니다.
    const bs=byBranch.get(r.branch);bs.days.add(r.date);
    // [학습] 기록 안의 각 상품을 확인합니다.
    for(const [name,item] of Object.entries(r.items)){
      // [학습] 수량·대체 마감 등 분석 제외 조건을 판정합니다.
      const status=stockStatus(item,r);
      // [학습] 아래 if/else는 한 상품·일을 상태 하나에만 분류합니다.
      if(status==="오픈 누락")quality.missingOpen++;
      else if(status==="마감 누락")quality.missingClose++;
      else if(status==="미입고/0")quality.noDelivery++;
      else if(status==="마감 대체")quality.estimated++;
      else if(status!=="정상")quality.invalid++;
      else quality.normal++;
      // [학습] 판매량 계산이 불가능하면 continue로 다음 상품으로 넘어갑니다.
      const sold=observedSales(item,r);if(sold===null)continue;
      // [학습] 정상 관측만 지점 판매·잔여 수량·유효 관측 수에 누적합니다. 잔여 수량을 폐기량으로 확정하지 않습니다.
      bs.sales+=sold;bs.remaining+=item.remaining;bs.valid++;
      // [학습] 처음 본 상품의 전체·평일·주말 누적값과 마감 0 횟수를 초기화합니다.
      if(!byProduct.has(name))byProduct.set(name,{name,total:0,days:0,wkTotal:0,wkN:0,weTotal:0,weN:0,stockouts:0});
      // [학습] 상품의 판매량과 유효 상품·일 수를 올립니다. 전체 지점에서는 같은 날짜도 지점별로 별도 관측입니다.
      const p=byProduct.get(name);p.total+=sold;p.days++;
      // [학습] 토·일인지에 따라 평일/주말 합계와 표본 수를 각각 누적합니다.
      if(weekend(r.date)){p.weTotal+=sold;p.weN++;}else{p.wkTotal+=sold;p.wkN++;}
      // [학습] 정상 관측 중 마감 0인 횟수를 셉니다.
      if(item.remaining===0)p.stockouts++;
    }
  }
  // [학습] Map을 화면용 배열로 바꾸고 상품은 총판매량 내림차순으로 정렬합니다. 지점의 날짜 수는 Set.size로 구합니다.
  return {byProduct:[...byProduct.values()].sort((a,b)=>b.total-a.total),byBranch:[...byBranch.entries()].map(([name,b])=>({name,days:b.days.size,sales:b.sales,remaining:b.remaining,valid:b.valid})),quality};
}
```

```js
// [학습] 기존 임시 생산량 추천식. 코드 이해 단계이므로 계수와 수식을 변경하지 않았습니다.
// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.
import { dayIndex, weekend } from './dates.js';
// [학습] import로 다른 파일의 기능을 가져옵니다. { ... } 안의 이름은 해당 파일이 export한 이름입니다.
import { canonicalRecord, observedSales } from './inventory.js';

// [학습] 원본의 임시 추천 규칙을 보존한 함수입니다. 팀원이 확정한 예측식이나 Newsvendor 모델이 아닙니다.
function predict(records,targetDate,names,products=[]){
  // [학습] 대상 요일과 여유 배수(평일 1.10, 주말 1.15)를 준비합니다.
  const wd=dayIndex(targetDate),buf=weekend(targetDate)?1.15:1.10;
  // [학습] 대상일보다 과거이면서 같은 요일인 기록만 남깁니다.
  const history=records.filter(r=>r.date<targetDate&&dayIndex(r.date)===wd)
    // [학습] 최신 날짜가 먼저 오도록 정렬하고 상품명을 표준화합니다.
    .sort((a,b)=>b.date.localeCompare(a.date)).map(r=>canonicalRecord(r,products));
  // [학습] 상품 이름마다 추천 결과 객체 하나를 만듭니다.
  return names.map(name=>{
    // [학습] 해당 상품의 유효 판매량 표본을 담을 배열입니다.
    const obs=[];
    // [학습] 최근 같은 요일의 기록부터 확인합니다.
    for(const r of history){
      // [학습] 이번 날짜에 해당 상품의 재고를 찾습니다.
      const item=r.items[name];
      // [학습] 정상 재고의 관측 판매량만 얻습니다.
      const s=observedSales(item,r);
      // [학습] 판매량 0도 유효 표본입니다. 정상 표본을 최대 8개 모으면 반복을 끝냅니다.
      if(s!==null){obs.push({sold:s,stockout:item.remaining===0});if(obs.length===8)break;}
    }
    // [학습] 자료가 없으면 추천량 null을 반환합니다. 0개 생산 추천과 구분합니다.
    if(!obs.length)return {name,qty:null,average:null,count:0,frequentStockout:false};
    // [학습] 최근 4개 표본은 가중치 2, 나머지는 1로 가중평균을 계산합니다. 분모도 실제 가중치 합입니다.
    const weighted=obs.reduce((acc,o,i)=>acc+o.sold*(i<4?2:1),0)/obs.reduce((acc,o,i)=>acc+(i<4?2:1),0);
    // [학습] 최근 최대 4개 중 마감 0 비율이 절반 이상인지 확인합니다.
    const recent=obs.slice(0,4),frequentStockout=recent.filter(o=>o.stockout).length/recent.length>=.5;
    // [학습] 가중평균에 여유 배수와 조건부 1.10을 곱하고 ceil로 올림합니다. average는 표시용 소수 첫째 자리 반올림입니다.
    return {name,qty:Math.ceil(weighted*buf*(frequentStockout?1.10:1)),average:Math.round(weighted*10)/10,count:obs.length,frequentStockout};
  });
}

// [학습] 이 파일의 함수를 외부에서 import할 수 있도록 공개합니다.
export { predict };
```

```jsx
  const scoped=useMemo(()=>branch===ALL?records:records.filter(r=>r.branch===branch),[branch,records]);

  // [학습] 검색어 앞뒤 공백을 제거하고 이름에 포함된 상품만 표에 표시합니다.

  const forecast=useMemo(()=>{

    if(branch!==ALL)return predictions[0]?.values||[];

      // [학습] 반영 지점이 있을 때만 합산합니다. 자료 없음은 0이 아니라 null입니다.
      return {name,qty:items.length?items.reduce((s,v)=>s+v.qty,0):null,
        // [학습] 표시용 average도 지점별 값을 합산합니다. 전체 지점의 단일 평균이라는 뜻은 아닙니다.
        average:items.length?items.reduce((s,v)=>s+v.average,0):null,
        // [학습] 전체 모드의 count는 표본 수가 아닌 반영 지점 수입니다. 어느 지점이든 마감 0 빈도가 높으면 태그를 켭니다.
        count:items.length,frequentStockout:items.some(v=>v.frequentStockout)};

  const reload=async()=>{

    const [r,p]=await Promise.all([loadRecords(),loadProducts()]);setRecords(r);setProducts(p);return r;

      .catch(e=>{if(mounted)setError("DB 로딩 실패: "+e.message);})
      // [학습] 성공·실패와 관계없이 최초 로딩 상태를 끝냅니다.
      .finally(()=>{if(mounted)setLoading(false);});
    // [학습] 화면에서 제거될 때 실행되는 정리 함수입니다. 요청 자체를 취소하지 않고 늦게 온 결과 반영을 막습니다.
    return()=>{mounted=false;};

  useEffect(()=>{

    const editable={};

    setInputs(editable);

  async function transact(action,success){

    setBusy(true);setError("");setMessage("");

    try{await action();await reload();setMessage(success);}
    // [학습] 오류를 표시하고 가능한 한 DB 상태를 다시 읽습니다. 현재 코드는 오류를 다시 throw하지 않는다는 점도 읽어두세요.
    catch(e){setError(e.message||String(e));try{await reload();}catch(_){/* 원래 오류 유지 */}}
    // [학습] 성공·실패 모두 처리 중 상태를 해제합니다.
    finally{setBusy(false);}
  }
  // [학습] 수동 입력 저장 버튼의 처리 함수입니다. 화면 입력값을 검증해 재고 객체로 만든 뒤 저장합니다.
  async function saveManual(){
    try{
      // [학습] 전체 지점에서는 입력·저장하지 않습니다. 개별 지점을 선택해야 합니다.

      const legacy=safeLegacy(selected,products);if(legacy)throw new Error(legacy);

      const items=structuredClone(selected?.items||{});

      for(const name of names){

        const entry=inputs[name]||{stock:"",remaining:""};

        const stock=optionalQty(entry.stock),remaining=optionalQty(entry.remaining);

        if(stock===null&&remaining===null){delete items[name];continue;}

      if(!window.confirm("이 날짜의 재고 입력값을 저장할까요? 기존 기록의 다른 보고 유형은 유지됩니다."))return;

  async function submitKakao(){
    // [학습] 결과가 없거나 검토 오류가 있으면 저장을 시작하지 않습니다.
    if(!parsed||parsed.issues.length)return;
    try{
      // [학습] 전체 지점에서는 입력·저장하지 않습니다. 개별 지점을 선택해야 합니다.
      if(branch===ALL)throw new Error("지점을 먼저 선택하세요.");
      // [학습] 파싱한 보고 날짜와 현재 선택 지점의 기존 기록을 찾습니다.
      const existing=records.find(r=>r.date===parsed.date&&r.branch===branch)||null;
      // [학습] 비표준 기존 기록의 원본 보존을 위해 수정 가능 여부를 확인합니다.
      const legacy=safeLegacy(existing,products);if(legacy)throw new Error(legacy);
      // [학습] 전체 교체/부분 반영 방식에 맞는 확인 문구를 만듭니다.
      const note=replaceType?"같은 유형의 이전 수량을 비우고 이번 보고로 대체합니다.":"이번 보고에 포함된 상품만 수정합니다.";
      // [학습] 브라우저 확인창입니다. 취소하면 현재 함수에서 즉시 돌아갑니다.
      if(!window.confirm(`${parsed.date} ${branch} ${parsed.type==="stock"?"오픈":"마감"} 보고를 저장할까요?\n${note}\n(기존 다른 유형의 재고는 유지)`))return;
      // [학습] 보고 결과에 branch를 추가하고 기존 재고와 합칩니다. 계산 단계에서는 아직 DB 저장을 하지 않습니다.
      const changed=updateReport(existing,{...parsed,branch},replaceType);
      // [학습] 새 상품 후보를 먼저 등록한 뒤 재고를 저장합니다. 두 요청은 하나의 원자적 작업이 아닙니다.
      await transact(async()=>{await addProducts(parsed.newProducts);await saveRecord(changed,existing);},"카카오톡 보고가 반영되었습니다.");
      // [학습] 처리 후 미리보기·원문을 비우고 보고 날짜로 화면을 이동합니다.
      setParsed(null);setPaste("");setDate(changed.date);setMode(null);
    // [학습] 수량 검증 등에서 생긴 예외를 화면 오류 상태에 기록합니다.
```

```js
// [학습] Node.js 기본 http 모듈로 파일 제공 서버를 만듭니다. 추가 패키지 설치는 필요하지 않습니다.
import http from 'node:http';
// [학습] fs/promises는 파일 읽기를 Promise와 await로 처리하는 API입니다.
import fs from 'node:fs/promises';
// [학습] path는 Windows 등 운영체제에 맞춰 파일 경로를 계산합니다.
import path from 'node:path';

// 로컬 학습·개발용 정적 서버. 실제 운영용 백엔드가 아니다.
// [학습] 명령을 실행한 프로젝트 폴더를 파일 제공 기준으로 사용합니다.
const root = path.resolve('.');
// [학습] 브라우저가 파일을 HTML·CSS·JavaScript로 해석하도록 확장자별 Content-Type을 지정합니다.
const mime = { '.html': 'text/html', '.css': 'text/css', '.js': 'text/javascript', '.jsx': 'text/plain' };
// [학습] 요청이 올 때마다 콜백을 실행합니다. request는 요청 정보, response는 돌려줄 응답입니다.
const server = http.createServer(async (request, response) => {
  try {
    // [학습] URL에서 경로를 꺼내고 한글 등 인코딩된 문자를 복원합니다.
    const pathname = decodeURIComponent(new URL(request.url, 'http://localhost').pathname);
    // [학습] / 요청은 index.html로 연결하며 상대 요청을 절대 파일 경로로 바꿉니다.
    const filename = path.resolve(root, '.' + (pathname === '/' ? '/index.html' : pathname));
    const relative = path.relative(root, filename);
    // [학습] 프로젝트 바깥 경로·숨김 파일·보관 원본에 대한 요청을 거부합니다. 로컬 개발용의 간단한 방어입니다.
    if (relative.startsWith('..') || path.isAbsolute(relative) || relative.split(path.sep).some(part => part.startsWith('.')) || relative.startsWith('reference' + path.sep)) {
      response.writeHead(403).end('Forbidden');
      return;
    }
    // [학습] 파일을 읽고 200 성공 상태와 형식을 응답합니다. no-store는 개발 중 오래된 파일 재사용을 줄입니다.
    const contents = await fs.readFile(filename);
    response.writeHead(200, { 'Content-Type': `${mime[path.extname(filename)] || 'text/plain'}; charset=utf-8`, 'Cache-Control': 'no-store' });
    // [학습] 실제 파일 내용을 브라우저에 전달하고 응답을 끝냅니다.
    response.end(contents);
  } catch {
    // [학습] 파일이 없거나 읽지 못하면 404 응답을 보냅니다.
    response.writeHead(404).end('Not found');
  }
});
// [학습] 이 컴퓨터에서만 접근 가능한 주소와 4173 포트로 실행합니다. 종료는 터미널 Ctrl+C입니다.
server.listen(4173, '127.0.0.1', () => console.log('시오사쿠 로컬 실행: http://127.0.0.1:4173'));
```
